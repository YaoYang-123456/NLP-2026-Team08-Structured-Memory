\begin{frame}{问题：长对话中的记忆不只是“存下来”}
    \large\key{智能体需要保留信息、跨会话找回它，并据此给出连贯回答。}\par
    \vspace{4mm}\normalsize
    \begin{itemize}\setlength{\itemsep}{4mm}
        \item 历史不断增长，相关信息可能分散在不同会话。
        \item 用户陈述的事实与智能体形成的判断容易混在一起。
        \item 只取相似片段，回答可能局部合理、整体矛盾。
    \end{itemize}
    \takeaway{Hindsight 把写入、检索和推理分成可检查的操作。}
    \note{约45秒。先交代问题范围：论文关注跨会话的长期对话记忆。作者把挑战分成保留、召回和推理。比如系统只检索到一条用户偏好，却没看到后来相反的说法，就可能给出前后不一致的回答。接下来按记忆结构和三种操作解释它如何处理这些问题。}
\end{frame}

\source{Hindsight, §3.1；PDF pp.2--3，表1}
\begin{frame}{四类网络把事实、摘要和意见分开}
    \small
    \begin{tabularx}{\linewidth}{@{}p{27mm}p{38mm}Y@{}}
        \toprule
        网络 & 记忆类型 & 例子或用途 \\
        \midrule
        World & 外部世界事实 & Alice 在 Google 的 Mountain View 团队工作。 \\
        Experience & 智能体自己的经历 & 智能体曾向 Alice 推荐徒步旅行。 \\
        Observation & 从事实归纳的实体摘要 & Alice 是专长于机器学习的软件工程师。 \\
        Opinion & 有置信度的主观判断 & “Python 适合数据科学”；置信度随证据更新。 \\
        \bottomrule
    \end{tabularx}
    \vspace{4mm}
    \takeaway{Observation 是证据摘要；Opinion 是带置信度的判断。}
    \aside{四类网络让开发者能检查智能体“知道什么”与“相信什么”。}
    \note{约55秒。World记录外部事实，Experience记录智能体自己的行动；这两类保存发生过什么。Observation是从多条事实综合出的实体画像，Opinion则明确是主观判断，并带有置信度。这个区分很重要：用户喜欢某种工具是证据，系统认为它适合用户则是判断。论文认为分开保存能提高可检查性，也为后续证据更新留出位置。}
\end{frame}

\begin{frame}{三种操作组成记忆生命周期}
    \centering
    \begin{tikzpicture}[
        >=Stealth,
        op/.style={draw=accent,rounded corners,align=center,text width=26mm,minimum height=20mm,inner sep=3pt,font=\small},
        bank/.style={draw=ink,rounded corners,align=center,text width=29mm,minimum height=20mm,inner sep=3pt,font=\small},
        flow/.style={->,thick,draw=accent}
    ]
        \node[op] (retain) { \textbf{Retain}\\对话／文档\\抽取、分类、标注时间};
        \node[bank,right=5mm of retain] (bank) {\textbf{四类记忆网络}\\World · Experience\\Observation · Opinion};
        \node[op,right=5mm of bank] (recall) {\textbf{Recall}\\查询＋token预算\\检索并排序};
        \node[op,right=5mm of recall] (reflect) {\textbf{Reflect}\\检索结果＋行为档案\\生成回答、更新意见};
        \draw[flow] (retain)--(bank);
        \draw[flow] (bank)--(recall);
        \draw[flow] (recall)--(reflect);
    \end{tikzpicture}
    \vspace{5mm}
    \aside{TEMPR 负责 retain／recall；CARA 负责 reflect。Reflect 的新证据可触发 Opinion 更新。}
    \note{约65秒。图中是系统主线。Retain把对话和文档转为结构化事实，标注类别、时间和实体关系；Recall根据问题与预算找回候选记忆；Reflect结合检索结果和行为档案生成回答，并能依据新证据更新Opinion。论文把前两步放在TEMPR组件中，把反思放在CARA中。下方说明强调新证据可以进入意见更新过程，但不代表每条记忆都必然改变已有意见。}
\end{frame}

\source{Hindsight, §3.2、§4.1；PDF pp.3、5（印刷页277、279）}
\begin{frame}{Recall 将四种检索并行，再融合排序}
    \[
        \{\text{向量},\ \text{关键词},\ \text{图遍历},\ \text{时间过滤}\}
        \ \longrightarrow\ \text{RRF}\ \longrightarrow\ \text{重排与预算裁剪}
    \]
    \vspace{2mm}
    \begin{tabularx}{\linewidth}{@{}p{29mm}Y@{}}
        \toprule
        通道 & 找什么 \\
        \midrule
        向量相似度 & 语义相近的记忆；使用 HNSW 索引。 \\
        BM25 & 关键词或实体名称；使用 PostgreSQL 全文检索。 \\
        图遍历 & 沿实体、时间、因果关系扩展，通常限制在 2--3 跳。 \\
        时间过滤 & 按问题中的日期或时间区间匹配事件。 \\
        \bottomrule
    \end{tabularx}
    \vspace{3mm}
    \key{RRF 合并各通道排序，cross-encoder 再重排；token 预算限制返回上下文。}
    \note{约55秒。四个通道互补：向量找语义相似，BM25找关键词，图遍历补实体和因果关联，时间过滤处理日期范围。它们并行运行后由RRF融合，再用cross-encoder重排。返回结果还受到token预算约束。论文称，对于一万条记忆，单次recall在不含回答模型推理时低于200毫秒；这个是作者报告的系统延迟，不是本组实测。}
\end{frame}

\source{Hindsight, §3.2--3.3、§6.4；PDF pp.3--4、7（印刷页277--278、281）}
\begin{frame}{Reflect 让意见随证据变化}
    \begin{columns}[T,onlytextwidth]
        \column{.47\textwidth}
        \textbf{行为档案}\par\vspace{3mm}
        \begin{itemize}\setlength{\itemsep}{3mm}
            \item 怀疑程度、字面倾向、共情程度
            \item 每项取值 1--5，用于调节回答倾向
            \item Opinion 保存置信度和支持证据
        \end{itemize}
        \column{.47\textwidth}
        \textbf{论文中的三会话示例}\par\vspace{3mm}
        \begin{itemize}\setlength{\itemsep}{3mm}
            \item 用户开始用 Rust 改造支付服务。
            \item 之后询问下一季度应重点做什么。
            \item 后来表示 Rust 难维护、更怀念 Go。
        \end{itemize}
    \end{columns}
    \vspace{3mm}
    \takeaway{新证据削弱先前的偏好判断；相关世界事实仍保留。}
    \aside{这是论文的定性案例，不是独立用户研究。}
    \note{约60秒。行为档案有怀疑、字面和共情三个维度，范围都是一到五，它们用于调节回答风格。意见有置信度，也能连接支持或反驳它的证据。论文的定性案例中，用户先说开始用Rust改造支付服务，之后咨询下一季度计划，后来又说Rust令人沮丧、怀念Go。新信息使对Rust偏好的观察由稳定转向减弱，但项目使用Rust这一事实仍然保留。}
\end{frame}

\begin{frame}{局限与证据边界}
    \begin{columns}[T,onlytextwidth]
        \column{.48\textwidth}
        \textbf{系统局限}\par\vspace{3mm}
        \begin{itemize}\setlength{\itemsep}{3mm}
            \item 抽取与实体消歧依赖 LLM，错误可能传播。
            \item 只评测英文数据；Opinion 置信度未做正式用户验证。
            \item 持续保存个人信息，且无内置 PII 自动脱敏。
        \end{itemize}
        \column{.48\textwidth}
        \textbf{报告证据需谨慎读}\par\vspace{3mm}
        \begin{itemize}\setlength{\itemsep}{3mm}
            \item 完整评估指向另一篇工作，本文引用编号仍是占位符。
            \item 正文与附录的部分类别基线数值不一致。
            \item 本文提供的是系统演示与结果摘要，不是本组复现。
        \end{itemize}
    \end{columns}
    \note{约60秒。作者自己指出了英文评测范围、模型抽取错误传播、意见质量缺少用户研究、隐私和脱敏等问题。证据方面，本文将完整评估指向另一篇2025年工作，但参考文献中的arXiv编号仍为占位符；另外，正文与附录的部分分类基线数字不同，文中没有充分解释。我的理解是，架构机制和作者报告的成绩可以作为阅读证据，但复述成绩时要保留它们的出处、条件和不确定之处。}
\end{frame}

\begin{frame}{对长上下文结构化记忆课设的启发}
    \begin{itemize}\setlength{\itemsep}{5mm}
        \item 分开事实、经历、实体摘要和意见，检查来源与置信度。
        \item 在固定 token 预算下，比较语义检索与时间／实体关联。
        \item 用消融区分各通道收益，并同步记录准确率、噪声和成本。
    \end{itemize}
    \vspace{4mm}
    \takeaway{可借鉴的是可检查的记忆类型与操作；迁移效果仍要用同一条件验证。}
    \aside{以上是个人阅读推导的实验问题，不是已经完成的本组实验或团队共识。}
    \note{约35秒。对我们的课设，值得转化为实验问题的是：记忆类型分开是否减少事实和意见混淆，时间与实体图能否补足向量检索，以及不同通道在固定预算下各有什么贡献。后续需要控制模型、数据和预算，并同时看准确率、错误记忆和成本。以上是我根据论文提出的设想，不代表我们已经做过这些实验，也不预设小组结论。}
\end{frame}
