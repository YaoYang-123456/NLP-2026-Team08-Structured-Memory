\begin{frame}{问题：经验既要保留细节，也要跟上任务变化}
    \large\key{截图与动作难以只靠文本总结；一次检索也会过时。}\par
    \vspace{5mm}\normalsize
    \begin{itemize}\setlength{\itemsep}{4mm}
        \item 输入：任务描述、当前截图、历史成功操作轨迹。
        \item 输出：指导智能体选择下一步网页操作。
        \item 例：搜索商品转入结账后，保留预算，更新操作指导。
    \end{itemize}
    \takeaway{混合表示解决“存什么”；自演化解决“何时更新”。}
    \note{约50秒。研究对象是网页操作智能体，评价任务成功率。搜索到结账是机制说明，不是本组实验。困难一是视觉细节难以完整文字化，二是任务状态变化使先前经验不再相关。}
\end{frame}

    \begin{columns}[T,onlytextwidth]

        \begin{tikzpicture}[node distance=5mm,box/.style={draw=accent,rounded corners,align=center,text width=53mm,minimum height=9mm,font=\small},>=Stealth]
            \node[box] (q) {任务文本＋当前截图};
            \node[box,below=of q] (r) {检索5条种子＋5条图邻居};
            \node[box,below=of r] (m) {离散指导指令＋连续轨迹嵌入};
            \node[box,below=of m] (a) {智能体执行下一步动作};
            \draw[->] (q)--(r);\draw[->] (r)--(m);\draw[->] (m)--(a);
        \end{tikzpicture}
        \column{.44\textwidth}
        \textbf{一条经验的三个层次}
        \[v_i=(c_i,A_i,m_i)\]
        \begin{itemize}\setlength{\itemsep}{3mm}
            \item $c_i$：可复用的高层策略。
            \item $A_i$：动作、界面及领域属性。
            \item $m_i$：8个连续轨迹嵌入。
        \end{itemize}
        \aside{CLIP检索索引与连续记忆是不同表示。}
    \end{columns}
    \note{约85秒。从任务与截图检索种子，再沿图关联扩展，最后生成文本指导并加入连续嵌入。5加5是主实验配置，不是理论最优。三元组描述经验内容，实际图存在策略、属性与轨迹的关联。CLIP负责找经验，CoMEM式编码负责让智能体使用轨迹细节。}
\end{frame}

\source{HyMEM，PDF 第4--5页，Memory Update、Working Memory Refresh；图1}
\begin{frame}{自演化：长期图与当前工作记忆分别更新}
    \begin{columns}[T,onlytextwidth]
        \column{.47\textwidth}
        \key{全局：跨任务积累经验}\par\vspace{3mm}
        \begin{itemize}\setlength{\itemsep}{3mm}
            \item ADD：出现新策略或属性。
            \item MERGE：同一策略的互补证据。
            \item REPLACE：同条件下更优的轨迹。
        \end{itemize}
        \aside{先检索候选，再由视觉语言模型判断信息增量。}
        \column{.47\textwidth}
        \key{局部：跟随当前任务阶段}\par\vspace{3mm}
        \begin{enumerate}\setlength{\itemsep}{3mm}
            \item 比较动作前后观测，检测阶段变化。
            \item 保留长期目标及约束，删除过时指导。
            \item 重新检索，刷新文本与连续嵌入。
        \end{enumerate}
    \end{columns}
    \takeaway{全局改变长期经验库；局部更新当前执行上下文。}
    \note{约85秒。相似不等于重复：新轨迹可能是补充证据，也可能是更优执行。局部在每次动作后判断是否换阶段，并非无条件重建全部记忆。两种更新都依赖模型判断，不能保证绝无误合并或误刷新。}
\end{frame}

\begin{frame}{阅读判断：借鉴组织机制，迁移收益仍需验证}
    \begin{columns}[T,onlytextwidth]
        \column{.47\textwidth}
        \textbf{贡献与局限}\par\vspace{3mm}
        \begin{itemize}\setlength{\itemsep}{3mm}
            \item 离散策略与视觉细节互补。
            \item 长期积累与任务内刷新形成闭环。
            \item 依赖训练、标签与模型判断，存在额外调用成本。
        \end{itemize}
        \column{.47\textwidth}
        \textbf{与本组StructMem D的联系}\par\vspace{3mm}
        \begin{itemize}\setlength{\itemsep}{3mm}
            \item 可研究种子检索与结构扩展的分工。
            \item 固定预算，比较覆盖、噪声及成本。
            \item 文本问答与截图操作口径不同，不能直接排名。
        \end{itemize}
    \end{columns}
    \aside{本组尚未复现HyMEM；迁移建议不是团队共识或已完成实验。}
    \note{约75秒。创新在于记忆组织与使用闭环，不能只记住35这个数字，忽略数据、训练和动作接口。我们的D部分做事件记忆综合与检索，可以提出固定预算下结构扩展是否改善覆盖的问题，但需要另做实验。交接语：以上是HyMEM的个人阅读分析，接下来进入下一篇论文。}
\end{frame}
